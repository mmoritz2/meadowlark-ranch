function ihash(ix,iz){
 let h=ix*374761393+iz*668265263;
 h=(h^(h>>13))*1274126177;
 return ((h^(h>>16))>>>0)/4294967295;
}
function vnoise(x,z){
 const ix=Math.floor(x),iz=Math.floor(z),fx=x-ix,fz=z-iz;
 const sx=fx*fx*(3-2*fx),sz=fz*fz*(3-2*fz);
 const a=ihash(ix,iz),b=ihash(ix+1,iz),c=ihash(ix,iz+1),d=ihash(ix+1,iz+1);
 return a+(b-a)*sx+(c-a)*sz+(a-b-c+d)*sx*sz;
}
const riverZ=x=>120+Math.sin(x*0.012)*45;
/* Sparrow Creek stream — a tributary that winds down from the southern hills into the river */
const streamX=z=>118+Math.sin(z*0.03)*14+Math.sin(z*0.011+3)*8;
const STREAM_JOIN=160;                          // it meets the river around z≈160
const riverLevel=x=>0.50-x*0.009;
const streamLevel=z=>THREE.MathUtils.lerp(12.5,riverLevel(streamX(STREAM_JOIN)),THREE.MathUtils.smoothstep(z,-330,STREAM_JOIN));
function rawH(x,z,fallsHeight=fallsTerrainHeight,canyonHeight=canyonRelief,riverHeight=riverBankTerrainHeight){
 const d=Math.hypot(x,z);
 const near=THREE.MathUtils.smoothstep(d,24,58);
 const far=1-THREE.MathUtils.smoothstep(d,380,460);
 const n=vnoise(x*0.018+7,z*0.018+3)*7.5       // medium rolling
        +vnoise(x*0.055,z*0.055)*1.8           // small bumps
        +vnoise(x*0.008+19,z*0.008+5)*14.0     // broad hills
        +vnoise(x*0.0042+50,z*0.0042+22)*13.0; // very broad ridges & valleys
 let h=near*far*(n-9.5)+pastureRise(x,z);
 const rg=1-Math.abs(vnoise(x*0.006+83,z*0.006-41)*2-1);   // ridged crests give hills a backbone
 h+=near*far*rg*rg*6.0;
 const bl=vnoise(x*0.011-31,z*0.011+44);                    // flat-topped bluffs, steep cliff faces
 h+=near*far*THREE.MathUtils.smoothstep(bl,0.35,0.60)*9.0;
 h+=canyonHeight(x,z);
 // A continuous graded channel: the water elevation is independent of the hills.
 const rd=Math.abs(z-riverZ(x));
 const riverBed=riverLevel(x)-0.85+THREE.MathUtils.smoothstep(rd,0,7)*1.30;
 h=THREE.MathUtils.lerp(riverBed,h,THREE.MathUtils.smoothstep(rd,7.5,19));
 h=riverHeight(x,z,h);
 if(z<STREAM_JOIN+3){
  const sd=Math.abs(x-streamX(z));
  const creekBed=streamLevel(Math.min(z,STREAM_JOIN))-.60+THREE.MathUtils.smoothstep(sd,0,3.5)*1.05;
  const creekInfluence=(1-THREE.MathUtils.smoothstep(sd,3.8,10))*(1-THREE.MathUtils.smoothstep(z,156,163));
  h=THREE.MathUtils.lerp(h,creekBed,creekInfluence);
 }

 return fallsHeight(x,z,oasisTerrainHeight(x,z,h));
}
